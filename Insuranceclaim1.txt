CREATE DATABASE claimssure_ai;
USE claimssure_ai;
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    phone VARCHAR(15),
    role ENUM('Policyholder','Claims Officer',
              'Surveyor','Admin') NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE TABLE policyholders (
    policyholder_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    address TEXT,
    city VARCHAR(100),
    state VARCHAR(100),
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);
CREATE TABLE policies (
    policy_id INT AUTO_INCREMENT PRIMARY KEY,
    policy_number VARCHAR(50) UNIQUE NOT NULL,
    policy_type VARCHAR(50),
    coverage_details TEXT,
    start_date DATE,
    end_date DATE,
    premium DECIMAL(12,2),
    coverage_amount DECIMAL(12,2),
    policy_status ENUM('Active','Expired','Cancelled')
);
CREATE TABLE vehicles (
    vehicle_id INT AUTO_INCREMENT PRIMARY KEY,
    policyholder_id INT NOT NULL,
    policy_id INT NOT NULL,
    vehicle_number VARCHAR(20) UNIQUE NOT NULL,
    vehicle_model VARCHAR(100),
    vehicle_type VARCHAR(50),
    manufacturing_year YEAR,
    FOREIGN KEY (policyholder_id)
        REFERENCES policyholders(policyholder_id),
    FOREIGN KEY (policy_id)
        REFERENCES policies(policy_id)
);

CREATE TABLE garages (
    garage_id INT AUTO_INCREMENT PRIMARY KEY,
    garage_name VARCHAR(150) NOT NULL,
    address TEXT,
    city VARCHAR(100),
    phone VARCHAR(15),
    email VARCHAR(100)
);

CREATE TABLE claims (
    claim_id INT AUTO_INCREMENT PRIMARY KEY,
    policyholder_id INT NOT NULL,
    vehicle_id INT NOT NULL,
    policy_id INT NOT NULL,
    garage_id INT,
    accident_date DATE,
    accident_description TEXT,
    claim_amount DECIMAL(12,2),
    claim_status ENUM(
        'Registered',
        'Under Review',
        'Survey Assigned',
        'Approved',
        'Rejected',
        'Settled'
    ) DEFAULT 'Registered',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (policyholder_id)
        REFERENCES policyholders(policyholder_id),
    FOREIGN KEY (vehicle_id)
        REFERENCES vehicles(vehicle_id),
    FOREIGN KEY (policy_id)
        REFERENCES policies(policy_id),
    FOREIGN KEY (garage_id)
        REFERENCES garages(garage_id)
);

CREATE TABLE claim_documents (
    document_id INT AUTO_INCREMENT PRIMARY KEY,
    claim_id INT NOT NULL,
    document_type VARCHAR(50),
    document_path VARCHAR(255),
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id)
);

CREATE TABLE survey_assignments (
    assignment_id INT AUTO_INCREMENT PRIMARY KEY,
    claim_id INT NOT NULL,
    surveyor_id INT NOT NULL,
    assigned_date DATE,
    assignment_status VARCHAR(50),
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id),
    FOREIGN KEY (surveyor_id) REFERENCES users(user_id)
);
CREATE TABLE inspection_reports (
    report_id INT AUTO_INCREMENT PRIMARY KEY,
    claim_id INT NOT NULL,
    surveyor_id INT NOT NULL,
    damage_description TEXT,
    inspection_date DATE,
    inspection_result VARCHAR(100),
    remarks TEXT,
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id),
    FOREIGN KEY (surveyor_id) REFERENCES users(user_id)
);
CREATE TABLE repair_estimates (
    estimate_id INT AUTO_INCREMENT PRIMARY KEY,
    claim_id INT NOT NULL,
    garage_id INT,
    parts_cost DECIMAL(12,2),
    labour_cost DECIMAL(12,2),
    total_estimated_cost DECIMAL(12,2),
    estimated_date DATE,
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id),
    FOREIGN KEY (garage_id) REFERENCES garages(garage_id)
);

CREATE TABLE fraud_assessments (
    assessment_id INT AUTO_INCREMENT PRIMARY KEY,
    claim_id INT NOT NULL,
    fraud_score DECIMAL(5,2),
    fraud_status ENUM('Low Risk','Medium Risk','High Risk'),
    fraud_reason TEXT,
    assessed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id)
);
CREATE TABLE settlements (
    settlement_id INT AUTO_INCREMENT PRIMARY KEY,
    claim_id INT NOT NULL,
    approved_amount DECIMAL(12,2),
    settlement_date DATE,
    payment_status ENUM('Pending','Processing','Completed','Failed'),
    settlement_remarks TEXT,
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id)
);
CREATE TABLE notifications (
    notification_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    claim_id INT,
    message TEXT,
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (claim_id) REFERENCES claims(claim_id)
);

SHOW TABLES;

# Important JOIN query #
# 1. Customer and Claim Details
SELECT
    u.name AS customer_name,
    c.claim_id,
    c.claim_amount,
    c.claim_status
FROM users u
JOIN policyholders p
    ON u.user_id = p.user_id
JOIN claims c
    ON p.policyholder_id = c.policyholder_id;
    # 2. Displays customer names and their vehicle information.
    SELECT
    u.name,
    v.vehicle_number,
    v.vehicle_model,
    v.vehicle_type
FROM users u
JOIN policyholders p
    ON u.user_id = p.user_id
JOIN vehicles v
    ON p.policyholder_id = v.policyholder_id;
    
   # 3. Displays claim amounts and AI fraud assessment results.
   SELECT
    c.claim_id,
    c.claim_amount,
    f.fraud_score,
    f.fraud_status,
    f.fraud_reason
FROM claims c
JOIN fraud_assessments f
    ON c.claim_id = f.claim_id;
    
    # 4. Displays which garage is handling each claim.
    SELECT
    c.claim_id,
    g.garage_name,
    g.city,
    c.claim_amount
FROM claims c
JOIN garages g
    ON c.garage_id = g.garage_id;
    
   # 5. Displays customer, vehicle, claim and fraud information together.
    SELECT
    u.name AS customer_name,
    v.vehicle_number,
    v.vehicle_model,
    c.claim_id,
    c.claim_amount,
    c.claim_status,
    f.fraud_score,
    f.fraud_status
FROM claims c
JOIN policyholders p
    ON c.policyholder_id = p.policyholder_id
JOIN users u
    ON p.user_id = u.user_id
JOIN vehicles v
    ON c.vehicle_id = v.vehicle_id
LEFT JOIN fraud_assessments f
    ON c.claim_id = f.claim_id;

CREATE VIEW fraud_details AS
SELECT
    c.claim_id,
    c.claim_amount,
    f.fraud_score,
    f.fraud_status,
    f.fraud_reason
FROM claims c
JOIN fraud_assessments f
ON c.claim_id = f.claim_id;
SELECT * FROM fraud_details;

# 5 Displays repair costs and garage information
CREATE VIEW repair_details AS
SELECT
    r.estimate_id,
    c.claim_id,
    g.garage_name,
    r.parts_cost,
    r.labour_cost,
    r.total_estimated_cost
FROM repair_estimates r
JOIN claims c
ON r.claim_id = c.claim_id
JOIN garages g
ON r.garage_id = g.garage_id;
SELECT * FROM repair_details;

# 6 Displays surveyor names and assigned claims.
CREATE VIEW surveyor_details AS

    s.assignment_id,
    c.claim_id,
    u.name AS surveyor_name,
    s.assigned_date,
    s.assignment_status

ON s.surveyor_id = u.user_id
JOIN claims c
ON s.claim_id = c.claim_id;
SELECT * FROM surveyor_details;
# 7 Displays customer settlement and payment information.
CREATE VIEW settlement_details AS

    s.claim_id,
    s.approved_amount,
    s.payment_status,
    s.settlement_date
FROM settlements s
JOIN claims c
ON s.claim_id = c.claim_id
JOIN policyholders p
ON c.policyholder_id = p.policyholder_id
JOIN users u
ON p.user_id = u.user_id;
SELECT * FROM settlement_details;

# Triggers #
#1 Prevent duplicate active claims for the same vehicle
USE claimssure_ai;

DELIMITER $$

CREATE TRIGGER trg_prevent_duplicate_claim
BEFORE INSERT ON claims
FOR EACH ROW
BEGIN
    IF EXISTS (
        SELECT 1
        FROM claims
        WHERE vehicle_id = NEW.vehicle_id
        AND claim_status IN (
            'Registered',
            'Under Review',
            'Survey Assigned',
            'Approved'
        )
    ) THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'An active claim already exists for this vehicle';
    END IF;
END $$

DELIMITER ;

#2 Prevent negative claim amounts
DELIMITER $$

CREATE TRIGGER trg_validate_claim_amount
BEFORE INSERT ON claims
FOR EACH ROW
BEGIN
    IF NEW.claim_amount <= 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Claim amount must be greater than zero';
    END IF;
END $$

DELIMITER ;
# 3 Validate fraud score
DELIMITER $$

CREATE TRIGGER trg_validate_fraud_score
BEFORE INSERT ON fraud_assessments
FOR EACH ROW
BEGIN
    IF NEW.fraud_score < 0 OR NEW.fraud_score > 100 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Fraud score must be between 0 and 100';
    END IF;
END $$

DELIMITER ;

#4 Automatically update claim status after successful payment
DELIMITER $$

CREATE TRIGGER trg_update_claim_after_settlement
AFTER UPDATE ON settlements
FOR EACH ROW
BEGIN
    IF NEW.payment_status = 'Completed'
       AND OLD.payment_status <> 'Completed' THEN

        UPDATE claims
        SET claim_status = 'Settled'
        WHERE claim_id = NEW.claim_id;

    END IF;
END $$

DELIMITER ;

# 5 Prevent settlement amount greater than claim amount
DELIMITER $$

CREATE TRIGGER trg_validate_settlement_amount
BEFORE INSERT ON settlements
FOR EACH ROW
BEGIN
    DECLARE v_claim_amount DECIMAL(12,2);

    SELECT claim_amount
    INTO v_claim_amount
    FROM claims
    WHERE claim_id = NEW.claim_id;

    IF NEW.approved_amount > v_claim_amount THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Settlement amount cannot exceed claim amount';
    END IF;
END $$

DELIMITER ;
